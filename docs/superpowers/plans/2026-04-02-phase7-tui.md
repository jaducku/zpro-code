// src/tui/app.rs
use std::io::Stdout;
use crossterm::event::{self, Event, KeyCode, KeyModifiers};
use ratatui::prelude::*;
use ratatui::Terminal;
use tokio::sync::mpsc;

use crate::query::{QueryContext, QueryEvent, query};
use crate::services::api::provider::LlmProvider;
use crate::tool::Tool;
use crate::types::content::ContentBlock;
use crate::types::message::{ApiMessage, Role};
use super::event::AppEvent;

/// Application state
pub struct App {
    /// Conversation messages for display
    pub messages: Vec<DisplayMessage>,
    /// Current input text
    pub input: String,
    /// Cursor position in input
    pub cursor_pos: usize,
    /// Whether the app should quit
    pub should_quit: bool,
    /// Whether a query is currently running
    pub is_loading: bool,
    /// Scroll offset for message list
    pub scroll_offset: u16,
    /// Status bar text
    pub status: String,
    /// Current model name
    pub model: String,
    /// Permission dialog (if showing)
    pub permission_dialog: Option<PermissionDialog>,
    /// API messages (conversation history for LLM)
    pub api_messages: Vec<ApiMessage>,
}

/// A message for display purposes
#[derive(Debug, Clone)]
pub struct DisplayMessage {
    pub role: DisplayRole,
    pub content: String,
    pub tool_name: Option<String>,
}

#[derive(Debug, Clone, PartialEq)]
pub enum DisplayRole {
    User,
    Assistant,
    System,
    Tool,
}

/// Permission dialog state
pub struct PermissionDialog {
    pub tool_name: String,
    pub description: String,
    pub selected: bool, // true = allow, false = deny
}

impl App {
    pub fn new(model: String) -> Self {
        Self {
            messages: Vec::new(),
            input: String::new(),
            cursor_pos: 0,
            should_quit: false,
            is_loading: false,
            scroll_offset: 0,
            status: format!("Model: {}", model),
            model,
            permission_dialog: None,
            api_messages: Vec::new(),
        }
    }

    /// Handle a key event
    pub fn handle_key(&mut self, key: KeyEvent) {
        if key.modifiers.contains(KeyModifiers::CONTROL) {
            match key.code {
                KeyCode::Char('c') | KeyCode::Char('d') => {
                    self.should_quit = true;
                }
                _ => {}
            }
            return;
        }

        match key.code {
            KeyCode::Char(c) => {
                self.input.insert(self.cursor_pos, c);
                self.cursor_pos += 1;
            }
            KeyCode::Backspace => {
                if self.cursor_pos > 0 {
                    self.cursor_pos -= 1;
                    self.input.remove(self.cursor_pos);
                }
            }
            KeyCode::Delete => {
                if self.cursor_pos < self.input.len() {
                    self.input.remove(self.cursor_pos);
                }
            }
            KeyCode::Left => {
                if self.cursor_pos > 0 {
                    self.cursor_pos -= 1;
                }
            }
            KeyCode::Right => {
                if self.cursor_pos < self.input.len() {
                    self.cursor_pos += 1;
                }
            }
            KeyCode::Home => {
                self.cursor_pos = 0;
            }
            KeyCode::End => {
                self.cursor_pos = self.input.len();
            }
            KeyCode::Up => {
                if self.scroll_offset > 0 {
                    self.scroll_offset -= 1;
                }
            }
            KeyCode::Down => {
                self.scroll_offset += 1;
            }
            KeyCode::Enter => {
                // Submit is handled by the main loop
            }
            _ => {}
        }
    }

    /// Take the current input (clears it)
    pub fn take_input(&mut self) -> String {
        let input = self.input.clone();
        self.input.clear();
        self.cursor_pos = 0;
        input
    }

    /// Add a display message
    pub fn push_message(&mut self, msg: DisplayMessage) {
        self.messages.push(msg);
    }

    /// Handle a query event (streaming update)
    pub fn handle_query_event(&mut self, event: QueryEvent) {
        match event {
            QueryEvent::TextDelta(text) => {
                if let Some(last) = self.messages.last_mut() {
                    if last.role == DisplayRole::Assistant {
                        last.content.push_str(&text);
                        return;
                    }
                }
                self.messages.push(DisplayMessage {
                    role: DisplayRole::Assistant,
                    content: text,
                    tool_name: None,
                });
            }
            QueryEvent::ToolCallStart { name, .. } => {
                self.messages.push(DisplayMessage {
                    role: DisplayRole::Tool,
                    content: format!("Running {}...", name),
                    tool_name: Some(name),
                });
            }
            QueryEvent::ToolCallComplete { result, .. } => {
                if let Some(last) = self.messages.last_mut() {
                    if last.role == DisplayRole::Tool {
                        last.content = result.content;
                    }
                }
            }
            QueryEvent::ResponseComplete { .. } => {
                self.is_loading = false;
            }
            QueryEvent::Error(msg) => {
                self.messages.push(DisplayMessage {
                    role: DisplayRole::System,
                    content: format!("Error: {}", msg),
                    tool_name: None,
                });
                self.is_loading = false;
            }
        }
    }
}

// src/tui/ui.rs
use ratatui::prelude::*;
use ratatui::widgets::*;

use super::app::{App, DisplayRole};

pub fn render(frame: &mut Frame, app: &App) {
    let layout = Layout::vertical([
        Constraint::Min(1),          // Messages
        Constraint::Length(1),       // Separator
        Constraint::Length(3),       // Input
        Constraint::Length(1),       // Status bar
    ]);
    let chunks = layout.split(frame.area());

    render_messages(frame, chunks[0], app);
    render_separator(frame, chunks[1]);
    render_input(frame, chunks[2], app);
    render_status_bar(frame, chunks[3], app);

    // Permission dialog overlay
    if let Some(dialog) = &app.permission_dialog {
        render_permission_dialog(frame, dialog);
    }
}

fn render_messages(frame: &mut Frame, area: Rect, app: &App) {
    let items: Vec<ListItem> = app
        .messages
        .iter()
        .map(|msg| {
            let (prefix, style) = match msg.role {
                DisplayRole::User => ("You", Style::default().fg(Color::Green).bold()),
                DisplayRole::Assistant => ("Assistant", Style::default().fg(Color::Cyan)),
                DisplayRole::System => ("System", Style::default().fg(Color::Yellow)),
                DisplayRole::Tool => {
                    let name = msg.tool_name.as_deref().unwrap_or("Tool");
                    return ListItem::new(Line::from(vec![
                        Span::styled(format!("[{}] ", name), Style::default().fg(Color::Magenta)),
                        Span::raw(truncate_content(&msg.content, area.width as usize - 20)),
                    ]));
                }
            };
            ListItem::new(Line::from(vec![
                Span::styled(format!("{}: ", prefix), style),
                Span::raw(truncate_content(&msg.content, area.width as usize - 15)),
            ]))
        })
        .collect();

    let messages_block = Block::default()
        .borders(Borders::NONE)
        .title(" Messages ");

    let list = List::new(items).block(messages_block);
    frame.render_widget(list, area);
}

fn render_separator(frame: &mut Frame, area: Rect) {
    let sep = Block::default()
        .borders(Borders::TOP)
        .border_style(Style::default().fg(Color::DarkGray));
    frame.render_widget(sep, area);
}

fn render_input(frame: &mut Frame, area: Rect, app: &App) {
    let prompt = if app.is_loading { " ... " } else { " > " };
    let input_block = Block::default()
        .borders(Borders::ALL)
        .border_style(if app.is_loading {
            Style::default().fg(Color::DarkGray)
        } else {
            Style::default().fg(Color::Blue)
        })
        .title(prompt);

    let input_text = Paragraph::new(app.input.as_str())
        .block(input_block);

    frame.render_widget(input_text, area);

    // Cursor position
    if !app.is_loading {
        frame.set_cursor_position((
            area.x + app.cursor_pos as u16 + 1,
            area.y + 1,
        ));
    }
}

fn render_status_bar(frame: &mut Frame, area: Rect, app: &App) {
    let status = Paragraph::new(app.status.as_str())
        .style(Style::default().fg(Color::White).bg(Color::DarkGray));
    frame.render_widget(status, area);
}

fn render_permission_dialog(frame: &mut Frame, dialog: &super::app::PermissionDialog) {
    let area = centered_rect(60, 30, frame.area());

    // Clear background
    frame.render_widget(Clear, area);

    let block = Block::default()
        .title(" Permission Required ")
        .borders(Borders::ALL)
        .border_style(Style::default().fg(Color::Yellow));

    let text = vec![
        Line::from(format!("Tool: {}", dialog.tool_name)),
        Line::from(""),
        Line::from(dialog.description.as_str()),
        Line::from(""),
        Line::from(vec![
            if dialog.selected {
                Span::styled("[Allow]", Style::default().fg(Color::Green).bold())
            } else {
                Span::styled(" Allow ", Style::default().fg(Color::DarkGray))
            },
            Span::raw("  "),
            if !dialog.selected {
                Span::styled("[Deny]", Style::default().fg(Color::Red).bold())
            } else {
                Span::styled(" Deny ", Style::default().fg(Color::DarkGray))
            },
        ]),
    ];

    let paragraph = Paragraph::new(text).block(block).wrap(Wrap { trim: false });
    frame.render_widget(paragraph, area);
}

fn truncate_content(content: &str, max_width: usize) -> String {
    let first_line = content.lines().next().unwrap_or("");
    if first_line.len() > max_width {
        format!("{}...", &first_line[..max_width.saturating_sub(3)])
    } else if content.lines().count() > 1 {
        format!("{} [+{} lines]", first_line, content.lines().count() - 1)
    } else {
        first_line.to_string()
    }
}

/// Helper to create a centered rectangle
fn centered_rect(percent_x: u16, percent_y: u16, area: Rect) -> Rect {
    let layout_v = Layout::vertical([
        Constraint::Percentage((100 - percent_y) / 2),
        Constraint::Percentage(percent_y),
        Constraint::Percentage((100 - percent_y) / 2),
    ]);
    let layout_h = Layout::horizontal([
        Constraint::Percentage((100 - percent_x) / 2),
        Constraint::Percentage(percent_x),
        Constraint::Percentage((100 - percent_x) / 2),
    ]);
    let v_chunks = layout_v.split(area);
    let h_chunks = layout_h.split(v_chunks[1]);
    h_chunks[1]
}

// src/tui/run.rs
use std::io;
use crossterm::{
    event::{self, Event, KeyCode},
    terminal::{disable_raw_mode, enable_raw_mode, EnterAlternateScreen, LeaveAlternateScreen},
    execute,
};
use ratatui::prelude::*;
use tokio::sync::mpsc;
use tokio_util::sync::CancellationToken;

use super::app::{App, DisplayMessage, DisplayRole};
use super::ui;
use crate::query::{QueryContext, QueryEvent, query};
use crate::services::api::provider::LlmProvider;
use crate::services::api::types::SystemBlock;
use crate::tool::Tool;
use crate::types::content::ContentBlock;
use crate::types::message::{ApiMessage, Role};

/// Run the TUI application
pub async fn run_app(
    provider: Box<dyn LlmProvider>,
    tools: Vec<Box<dyn Tool>>,
    system_prompt: Option<Vec<SystemBlock>>,
) -> io::Result<()> {
    // Setup terminal
    enable_raw_mode()?;
    let mut stdout = io::stdout();
    execute!(stdout, EnterAlternateScreen)?;
    let backend = CrosstermBackend::new(stdout);
    let mut terminal = Terminal::new(backend)?;

    let model = provider.model_info().id.clone();
    let mut app = App::new(model.clone());

    let (query_event_tx, mut query_event_rx) = mpsc::unbounded_channel::<QueryEvent>();

    loop {
        // Render
        terminal.draw(|frame| ui::render(frame, &app))?;

        // Handle events with timeout for non-blocking
        tokio::select! {
            // Terminal events (keyboard, mouse, resize)
            _ = async {
                if event::poll(std::time::Duration::from_millis(50)).unwrap_or(false) {
                    if let Ok(ev) = event::read() {
                        match ev {
                            Event::Key(key) => {
                                if key.code == KeyCode::Enter && !app.is_loading && !app.input.is_empty() {
                                    let input = app.take_input();
                                    submit_query(
                                        &mut app,
                                        &input,
                                        &*provider,
                                        &tools,
                                        &system_prompt,
                                        &model,
                                        query_event_tx.clone(),
                                    ).await;
                                } else {
                                    app.handle_key(key);
                                }
                            }
                            Event::Resize(w, h) => {
                                app.status = format!("Model: {} | {}x{}", model, w, h);
                            }
                            _ => {}
                        }
                    }
                }
            } => {}

            // Query events (streaming responses)
            Some(event) = query_event_rx.recv() => {
                app.handle_query_event(event);
            }
        }

        if app.should_quit {
            break;
        }
    }

    // Restore terminal
    disable_raw_mode()?;
    execute!(terminal.backend_mut(), LeaveAlternateScreen)?;
    terminal.show_cursor()?;

    Ok(())
}

async fn submit_query(
    app: &mut App,
    input: &str,
    provider: &dyn LlmProvider,
    tools: &[Box<dyn Tool>],
    system_prompt: &Option<Vec<SystemBlock>>,
    model: &str,
    event_tx: mpsc::UnboundedSender<QueryEvent>,
) {
    // Add user message to display
    app.push_message(DisplayMessage {
        role: DisplayRole::User,
        content: input.to_string(),
        tool_name: None,
    });

    // Add to API messages
    app.api_messages.push(ApiMessage {
        role: Role::User,
        content: vec![ContentBlock::Text { text: input.to_string() }],
    });

    app.is_loading = true;

    // Note: In a full implementation, query() would run in a spawned task
    // For now, it runs inline (blocking the event loop during execution)
    let ctx = QueryContext {
        messages: app.api_messages.clone(),
        tools: Vec::new(), // Tools can't be cloned; would need Arc in production
        system_prompt: system_prompt.clone(),
        model: model.to_string(),
        max_tokens: 4096,
    };

    // The actual query execution would be spawned as a background task
    // For the skeleton, we just set loading state
    // tokio::spawn(async move { query(ctx, provider, event_tx, cancel).await });
    app.status = format!("Model: {} | Thinking...", model);
}

// src/main.rs
use clap::Parser;

#[derive(Parser)]
#[command(name = "atom-code", version)]
struct Cli {
    /// Initial prompt (non-interactive mode)
    prompt: Option<String>,

    /// Model to use
    #[arg(long, default_value = "default")]
    model: Option<String>,

    /// API base URL
    #[arg(long, env = "ATOM_API_BASE_URL")]
    api_base: Option<String>,

    /// API key
    #[arg(long, env = "ATOM_API_KEY")]
    api_key: Option<String>,
}

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    let cli = Cli::parse();

    // Display version if no interactive mode
    if cli.prompt.is_none() && cli.api_base.is_none() {
        println!("atom-code v{}", env!("CARGO_PKG_VERSION"));
        println!("Use --api-base and --api-key to connect to an LLM provider");
        return Ok(());
    }

    println!("atom-code v{}", env!("CARGO_PKG_VERSION"));
    Ok(())
}
